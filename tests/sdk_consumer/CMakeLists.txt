cmake_minimum_required(VERSION 3.28)
project(NgenSdkConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
if(WITH_RENDERER)
  find_package(NGEN CONFIG REQUIRED COMPONENTS Renderer PATHS "${SDK_PREFIX}" NO_DEFAULT_PATH)
  add_executable(consumer renderer.cpp)
  target_link_libraries(consumer PRIVATE NGEN::renderer)
else()
  set(CMAKE_DISABLE_FIND_PACKAGE_SDL3 TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_Vulkan TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_VulkanMemoryAllocator TRUE)
  find_package(NGEN CONFIG REQUIRED PATHS "${SDK_PREFIX}" NO_DEFAULT_PATH)
  if(TARGET NGEN::renderer OR TARGET SDL3::SDL3 OR TARGET Vulkan::Vulkan)
    message(FATAL_ERROR "Headless SDK discovery imported a graphics dependency")
  endif()
  add_executable(consumer headless.cpp)
  target_link_libraries(consumer PRIVATE NGEN::simulation)
endif()
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(consumer PRIVATE /W4 /WX)
else()
  target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()
add_test(NAME sdk_consumer COMMAND consumer)
set_tests_properties(sdk_consumer PROPERTIES TIMEOUT 30)
