cmake_minimum_required(VERSION 3.28)
project(NgenM4ContactProbe LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(Jolt 5.6 CONFIG REQUIRED)
set(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../../engine")
add_executable(contact_probe contact_probe.cpp "${engine}/physics/world.cpp" "${engine}/core/values.cpp")
target_include_directories(contact_probe PRIVATE "${engine}")
target_compile_features(contact_probe PRIVATE cxx_std_20)
target_compile_options(contact_probe PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
target_link_libraries(contact_probe PRIVATE Jolt::Jolt Threads::Threads)
