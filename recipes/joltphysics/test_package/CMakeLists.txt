cmake_minimum_required(VERSION 3.28)
project(headless_double LANGUAGES CXX)
find_package(Jolt CONFIG REQUIRED)
add_executable(headless_double main.cpp)
target_compile_features(headless_double PRIVATE cxx_std_20)
set_target_properties(headless_double PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(headless_double PRIVATE Jolt::Jolt)
if(MSVC)
  target_compile_options(headless_double PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(headless_double PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()
