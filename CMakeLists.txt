cmake_minimum_required(VERSION 3.28)
project(NGEN VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
find_package(enkiTS CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(NGEN_SANITIZER "" CACHE STRING "Sanitizer: address,undefined or thread")
add_library(ngen_options INTERFACE)
target_compile_features(ngen_options INTERFACE cxx_std_20)
set(CMAKE_CXX_EXTENSIONS OFF)
target_compile_options(ngen_options INTERFACE
  $<$<CXX_COMPILER_ID:Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wsign-conversion;-Werror>
  $<$<CXX_COMPILER_ID:MSVC>:/W4;/WX;/permissive->)
if(NGEN_SANITIZER)
  if(NOT NGEN_SANITIZER MATCHES "^(address,undefined|thread)$" OR MSVC)
    message(FATAL_ERROR "Use address,undefined or thread with a supported Linux compiler")
  endif()
  target_compile_options(ngen_options INTERFACE -fsanitize=${NGEN_SANITIZER} -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(ngen_options INTERFACE -fsanitize=${NGEN_SANITIZER} -fno-sanitize-recover=all)
endif()
add_library(ngen_core engine/core/values.cpp engine/core/entities.cpp)
target_include_directories(ngen_core PUBLIC engine)
target_link_libraries(ngen_core PUBLIC ngen_options)
add_library(ngen_jobs engine/jobs/jobs.cpp)
target_link_libraries(ngen_jobs PUBLIC ngen_core enkiTS::enkiTS Threads::Threads)
add_library(ngen_simulation engine/simulation/simulation.cpp)
target_link_libraries(ngen_simulation PUBLIC ngen_jobs)
add_executable(sim_harness apps/sim_harness/main.cpp)
target_link_libraries(sim_harness PRIVATE ngen_simulation)
if(NGEN_SANITIZER)
  target_compile_definitions(sim_harness PRIVATE NGEN_ALLOCATION_PROBE=0)
else()
  target_compile_definitions(sim_harness PRIVATE NGEN_ALLOCATION_PROBE=1)
endif()
if(BUILD_TESTING)
  find_package(Catch2 3 CONFIG REQUIRED)
  add_executable(ngen_tests tests/core.cpp tests/jobs.cpp tests/simulation.cpp)
  target_link_libraries(ngen_tests PRIVATE ngen_simulation Catch2::Catch2WithMain)
  add_test(NAME ngen_tests COMMAND ngen_tests)
  add_test(NAME replay_serial COMMAND sim_harness)
  add_test(NAME replay_workers COMMAND sim_harness --workers=2)
  set_tests_properties(ngen_tests replay_serial replay_workers PROPERTIES TIMEOUT 60)
endif()
