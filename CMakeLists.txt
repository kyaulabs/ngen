cmake_minimum_required(VERSION 3.28)
project(NGEN VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
find_package(enkiTS CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(NGEN_SANITIZER "" CACHE STRING "Sanitizer: address,undefined or thread")
add_library(ngen_options INTERFACE)
target_compile_features(ngen_options INTERFACE cxx_std_20)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(ngen_options INTERFACE /W4)
else()
  target_compile_options(ngen_options INTERFACE $<$<CXX_COMPILER_ID:Clang,GNU>:-Wall>)
endif()
target_compile_options(ngen_options INTERFACE
  $<$<CXX_COMPILER_ID:Clang,GNU>:-Wextra;-Wpedantic;-Wconversion;-Wsign-conversion;-Werror>
  $<$<CXX_COMPILER_ID:MSVC>:/WX;/permissive->)
if(NGEN_SANITIZER)
  if(NOT NGEN_SANITIZER MATCHES "^(address,undefined|thread)$" OR MSVC)
    message(FATAL_ERROR "Use address,undefined or thread with a supported Linux compiler")
  endif()
  target_compile_options(ngen_options INTERFACE -fsanitize=${NGEN_SANITIZER} -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(ngen_options INTERFACE -fsanitize=${NGEN_SANITIZER} -fno-sanitize-recover=all)
endif()
add_library(ngen_core engine/core/values.cpp engine/core/entities.cpp)
target_include_directories(ngen_core PUBLIC engine)
target_link_libraries(ngen_core PUBLIC ngen_options)
add_library(ngen_jobs engine/jobs/jobs.cpp)
target_link_libraries(ngen_jobs PUBLIC ngen_core enkiTS::enkiTS Threads::Threads)
add_library(ngen_simulation engine/simulation/simulation.cpp)
target_link_libraries(ngen_simulation PUBLIC ngen_jobs)
add_executable(sim_harness apps/sim_harness/main.cpp)
target_link_libraries(sim_harness PRIVATE ngen_simulation)
if(NGEN_SANITIZER)
  target_compile_definitions(sim_harness PRIVATE NGEN_ALLOCATION_PROBE=0)
else()
  target_compile_definitions(sim_harness PRIVATE NGEN_ALLOCATION_PROBE=1)
endif()
add_library(ngen_render_contract engine/render/capabilities.cpp)
target_include_directories(ngen_render_contract PUBLIC engine)
target_link_libraries(ngen_render_contract PUBLIC ngen_options)
option(NGEN_BUILD_ASSETS "Build the checked runtime scene package loader" OFF)
if(NGEN_BUILD_ASSETS)
  find_package(ZLIB 1.3.2 CONFIG REQUIRED)
  add_library(ngen_assets engine/assets/package.cpp)
  target_include_directories(ngen_assets PUBLIC engine)
  target_link_libraries(ngen_assets PUBLIC ngen_options PRIVATE ZLIB::ZLIB)
  add_executable(asset_package_probe apps/asset_probe/main.cpp)
  target_link_libraries(asset_package_probe PRIVATE ngen_assets)
endif()
option(NGEN_BUILD_ASSET_TOOLS "Build the offline static glTF importer" OFF)
if(NGEN_BUILD_ASSET_TOOLS)
  find_package(fastgltf 0.9.1 CONFIG REQUIRED)
  find_package(simdjson 4.6.4 CONFIG REQUIRED)
  find_package(PNG 1.6.59 CONFIG REQUIRED)
  find_package(mikktspace CONFIG REQUIRED)
  find_package(meshoptimizer 1.0 CONFIG REQUIRED)
  add_library(ngen_asset_import tools/assetc/import.cpp tools/assetc/metadata.cpp tools/assetc/png.cpp tools/assetc/tangents.cpp)
  target_include_directories(ngen_asset_import PUBLIC tools)
  target_link_libraries(ngen_asset_import PUBLIC ngen_options PRIVATE fastgltf::fastgltf simdjson::simdjson PNG::PNG mikktspace::mikktspace meshoptimizer::meshoptimizer)
  add_library(ngen_asset_cook tools/assetc/cook.cpp tools/assetc/stage.cpp tools/assetc/source.cpp)
  target_link_libraries(ngen_asset_cook PUBLIC ngen_asset_import ngen_assets PRIVATE meshoptimizer::meshoptimizer)
  add_executable(assetc tools/assetc/main.cpp)
  target_link_libraries(assetc PRIVATE ngen_asset_cook)
endif()
option(NGEN_BUILD_RENDERER "Build the Vulkan/SDL startup path" OFF)
if(NGEN_BUILD_RENDERER)
  find_package(SDL3 3.4 CONFIG REQUIRED)
  find_package(Vulkan 1.3 REQUIRED)
  find_package(VulkanMemoryAllocator 3.3 CONFIG REQUIRED)
  add_library(ngen_vma engine/render/vma.cpp)
  target_link_libraries(ngen_vma PUBLIC GPUOpen::VulkanMemoryAllocator PRIVATE ngen_options Vulkan::Vulkan)
  target_compile_definitions(ngen_vma PUBLIC VMA_VULKAN_VERSION=1003000)
  add_library(ngen_render engine/render/context.cpp engine/render/surface.cpp engine/render/presentation.cpp engine/render/resources.cpp)
  target_link_libraries(ngen_render PUBLIC ngen_render_contract SDL3::SDL3 Vulkan::Vulkan PRIVATE ngen_vma)
  add_library(ngen_probe_support apps/probe/presentation.cpp)
  target_link_libraries(ngen_probe_support PUBLIC ngen_render)
  target_include_directories(ngen_probe_support PUBLIC apps)
  add_executable(render_probe apps/render_probe/main.cpp)
  target_link_libraries(render_probe PRIVATE ngen_probe_support)
  add_executable(resource_probe apps/resource_probe/main.cpp)
  target_link_libraries(resource_probe PRIVATE ngen_probe_support)
endif()
if(BUILD_TESTING)
  find_package(Catch2 3 CONFIG REQUIRED)
  add_executable(ngen_tests tests/core.cpp tests/jobs.cpp tests/simulation.cpp)
  target_link_libraries(ngen_tests PRIVATE ngen_simulation Catch2::Catch2WithMain)
  add_test(NAME ngen_tests COMMAND ngen_tests)
  add_executable(render_contract_tests tests/render_capabilities.cpp)
  target_link_libraries(render_contract_tests PRIVATE ngen_render_contract Catch2::Catch2WithMain)
  add_test(NAME render_contract_tests COMMAND render_contract_tests)
  if(NGEN_BUILD_ASSETS)
    add_executable(asset_package_tests tests/assets.cpp)
    target_link_libraries(asset_package_tests PRIVATE ngen_assets Catch2::Catch2WithMain ZLIB::ZLIB)
    target_compile_definitions(asset_package_tests PRIVATE NGEN_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME asset_package_tests COMMAND asset_package_tests)
    set_tests_properties(asset_package_tests PROPERTIES TIMEOUT 90)
  endif()
  if(NGEN_BUILD_ASSET_TOOLS)
    add_executable(asset_import_tests tests/asset_import.cpp)
    target_link_libraries(asset_import_tests PRIVATE ngen_asset_import Catch2::Catch2WithMain)
    add_test(NAME asset_import_tests COMMAND asset_import_tests)
    add_executable(asset_cook_tests tests/asset_cook.cpp)
    target_link_libraries(asset_cook_tests PRIVATE ngen_asset_cook Catch2::Catch2WithMain)
    add_test(NAME asset_cook_tests COMMAND asset_cook_tests)
    set_tests_properties(asset_cook_tests PROPERTIES TIMEOUT 90)
    set_tests_properties(asset_import_tests PROPERTIES TIMEOUT 90)
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    add_test(NAME asset_cli COMMAND ${CMAKE_COMMAND} -E env
      "NGEN_ASSETC_EXE=$<TARGET_FILE:assetc>"
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/asset_cli.py)
    set_tests_properties(asset_cli PROPERTIES TIMEOUT 90)
    add_test(NAME asset_cook_cli COMMAND ${CMAKE_COMMAND} -E env
      "NGEN_ASSETC_EXE=$<TARGET_FILE:assetc>"
      "NGEN_KTX_EXE=${NGEN_KTX_EXECUTABLE}"
      "NGEN_PACKAGE_PROBE=$<TARGET_FILE:asset_package_probe>"
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/asset_cooking.py)
    set_tests_properties(asset_cook_cli PROPERTIES TIMEOUT 180)
  endif()
  if(NGEN_BUILD_RENDERER)
    add_executable(render_startup_tests tests/render_startup.cpp tests/render_surface.cpp)
    target_link_libraries(render_startup_tests PRIVATE ngen_render Catch2::Catch2WithMain)
    add_test(NAME render_startup_tests COMMAND render_startup_tests)
    set_tests_properties(render_startup_tests PROPERTIES TIMEOUT 30)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      find_package(wayland CONFIG REQUIRED)
      add_executable(wayland_tablet_tests tests/wayland_tablets.cpp)
      target_link_libraries(wayland_tablet_tests PRIVATE ngen_options SDL3::SDL3 wayland::wayland-server wayland::wayland-client)
      add_test(NAME wayland_tablet_shutdown COMMAND wayland_tablet_tests)
      add_test(NAME wayland_tablet_removal COMMAND wayland_tablet_tests --remove)
      set_tests_properties(wayland_tablet_shutdown wayland_tablet_removal PROPERTIES TIMEOUT 20)
    endif()
  endif()
  add_test(NAME replay_serial COMMAND sim_harness)
  add_test(NAME replay_workers COMMAND sim_harness --workers=2)
  set_tests_properties(ngen_tests replay_serial replay_workers PROPERTIES TIMEOUT 60)
endif()
