cmake_minimum_required(VERSION 3.28)
project(NGENStarter LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(NGEN CONFIG REQUIRED COMPONENTS Assets)
set(NGEN_STARTER_SCENE "" CACHE FILEPATH "Cooked scene.ngas produced by the NGEN cooker")
if(NOT EXISTS "${NGEN_STARTER_SCENE}" OR IS_DIRECTORY "${NGEN_STARTER_SCENE}")
  message(FATAL_ERROR "Set NGEN_STARTER_SCENE to the checked scene.ngas from the offline cooker")
endif()
add_executable(ngen_starter main.cpp)
target_link_libraries(ngen_starter PRIVATE NGEN::simulation NGEN::assets)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(ngen_starter PRIVATE /W4 /WX)
else()
  target_compile_options(ngen_starter PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()
enable_testing()
add_test(NAME starter COMMAND ngen_starter "${NGEN_STARTER_SCENE}")
set_tests_properties(starter PROPERTIES TIMEOUT 30)
